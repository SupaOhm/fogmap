# M1 Plan 1: Scaffold and Core Logic Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** An Expo SDK 58 TypeScript app that builds on a physical iPhone with h3-js verified under Hermes, plus every pure function in `src/core/` (filtering, gap filling, reveal pipeline, stats, fog geometry, GPX parsing, stress data) with Jest tests.

**Tech Stack:** Expo SDK 58, React Native (Hermes), TypeScript (strict), h3-js 4.x, Jest via jest-expo.

**Spec:** [docs/superpowers/specs/2026-10-28-milestone-1-design.md](../specs/2026-10-28-milestone-1-design.md)

```ts
import {
  UNITS,
  cellArea,
  cellsToMultiPolygon,
  gridDisk,
  gridPathCells,
  isValidCell,
  latLngToCell,
} from 'h3-js';
import { H3_RES } from './constants';

export type CheckResult = { name: string; ok: boolean; detail: string };

const BANGKOK = { lat: 13.7563, lng: 100.5018 };

function assert(condition: boolean, message: string): void {
  if (!condition) throw new Error(message);
}

/** Exercises every h3-js function the app uses; safe to run on device. */
export function runH3SelfCheck(): CheckResult[] {
  const origin = latLngToCell(BANGKOK.lat, BANGKOK.lng, H3_RES);
  const checks: Array<[string, () => string]> = [
    ['latLngToCell', () => {
      assert(isValidCell(origin), `invalid cell ${origin}`);
      return origin;
    }],
    ['gridDisk', () => {
      const n = gridDisk(origin, 1).length;
      assert(n === 7, `expected 7 cells, got ${n}`);
      return `${n} cells`;
    }],
    ['gridPathCells', () => {
      const end = latLngToCell(BANGKOK.lat, BANGKOK.lng + 0.01, H3_RES);
      const path = gridPathCells(origin, end);
      assert(path.length > 2, `path too short: ${path.length}`);
      return `${path.length} cells over ~1 km`;
    }],
    ['cellsToMultiPolygon', () => {
      const cells = gridDisk(origin, 50);
      const t0 = Date.now();
      const polygons = cellsToMultiPolygon(cells, true);
      const ms = Date.now() - t0;
      assert(polygons.length === 1, `expected 1 polygon, got ${polygons.length}`);
      return `${cells.length} cells -> ${polygons.length} polygon in ${ms} ms`;
    }],
    ['cellArea', () => {
      const km2 = cellArea(origin, UNITS.km2);
      assert(km2 > 0.01 && km2 < 0.02, `unexpected area ${km2}`);
      return `${km2.toFixed(5)} km²`;
    }],
  ];
  return checks.map(([name, run]) => {
    try {
      return { name, ok: true, detail: run() };
    } catch (e) {
      return { name, ok: false, detail: e instanceof Error ? e.message : String(e) };
    }
  });
}
```

```ts
export type Fix = {
  ts: number;
  lat: number;
  lng: number;
  accuracy: number | null;
  speed: number | null;
};

export type SpeedBucket = 'walk' | 'cycle' | 'vehicle';

export type LogRow = {
  ts: number;
  accuracy: number | null;
  speed: number | null;
  accepted: boolean;
  rejectReason: RejectReason | null;
};

export type TrackingMode = 'efficient' | 'precise';
```

```ts
import { CYCLE_MAX_KMH, WALK_MAX_KMH } from './constants';
import type { SpeedBucket } from './types';

const ORDER: Record<SpeedBucket, number> = { walk: 0, cycle: 1, vehicle: 2 };

export function speedBucket(kmh: number): SpeedBucket {
  if (kmh <= WALK_MAX_KMH) return 'walk';
  if (kmh <= CYCLE_MAX_KMH) return 'cycle';
  return 'vehicle';
}

/** A cell keeps the slowest mode it was ever revealed in. */
export function slowerBucket(a: SpeedBucket, b: SpeedBucket): SpeedBucket {
  return ORDER[a] <= ORDER[b] ? a : b;
}
```

```ts
import { UNITS, cellToBoundary, cellToLatLng, greatCircleDistance } from 'h3-js';
import type { LastPoint } from './types';

export function distanceM(aLat: number, aLng: number, bLat: number, bLng: number): number {
  return greatCircleDistance([aLat, aLng], [bLat, bLng], UNITS.m);
}

/** Largest distance from the cell centre to any of its vertices. */
export function cellRadiusM(cell: string): number {
  const [lat, lng] = cellToLatLng(cell);
  let max = 0;
  for (const [vLat, vLng] of cellToBoundary(cell)) {
    max = Math.max(max, distanceM(lat, lng, vLat, vLng));
  }
  return max;
}

/**
 * Distance from the previous point. With only a cell (after a relaunch) this is a
 * lower bound, so imprecision can never make a real fix look too fast.
 */
export function distanceFromPrevM(prev: LastPoint, lat: number, lng: number): number {
  if (prev.lat !== undefined && prev.lng !== undefined) {
    return distanceM(prev.lat, prev.lng, lat, lng);
  }
  const [cLat, cLng] = cellToLatLng(prev.cell);
  return Math.max(0, distanceM(cLat, cLng, lat, lng) - cellRadiusM(prev.cell));
}
```

```ts
const LAT = 13.7563;
const LNG = 100.5018;
const EAST_1KM = LNG + 0.01; // ~1081 m east

function fix(over: Partial<Fix> = {}): Fix {
  return { ts: 0, lat: LAT, lng: LNG, accuracy: 10, speed: null, ...over };
}

function exactPrev(ts: number): LastPoint {
  return { cell: latLngToCell(LAT, LNG, H3_RES), ts, lat: LAT, lng: LNG };
}

describe('accuracy', () => {
  test('rejects missing accuracy', () => {
    expect(evaluateFix(fix({ accuracy: null }), null)).toEqual({ accepted: false, reason: 'accuracy' });
  });
  test('rejects accuracy worse than 50 m', () => {
    expect(evaluateFix(fix({ accuracy: 51 }), null)).toEqual({ accepted: false, reason: 'accuracy' });
  });
  test('accepts accuracy of exactly 50 m', () => {
    expect(evaluateFix(fix({ accuracy: 50 }), null).accepted).toBe(true);
  });
});

describe('first fix (no previous point)', () => {
  test('without speed is bucketed as walk', () => {
    expect(evaluateFix(fix(), null)).toEqual({ accepted: true, bucket: 'walk' });
  });
  test('uses reported speed', () => {
    expect(evaluateFix(fix({ speed: 10 }), null)).toEqual({ accepted: true, bucket: 'vehicle' });
  });
});

describe('speed check', () => {
  test('accepts ~65 km/h (1081 m in 60 s)', () => {
    expect(evaluateFix(fix({ ts: 60_000, lng: EAST_1KM }), exactPrev(0))).toEqual({
```

```ts
import { speedBucket } from './buckets';
import { MAX_ACCURACY_M, MAX_SPEED_KMH } from './constants';
import { distanceFromPrevM } from './geo';
import type { Fix, FixVerdict, LastPoint } from './types';

function reportedSpeedKmh(fix: Fix): number | null {
  return fix.speed !== null && fix.speed >= 0 ? fix.speed * 3.6 : null;
}

export function evaluateFix(fix: Fix, prev: LastPoint | null): FixVerdict {
  if (fix.accuracy === null || fix.accuracy > MAX_ACCURACY_M) {
    return { accepted: false, reason: 'accuracy' };
  }
  let computedKmh: number | null = null;
  if (prev !== null) {
    // Clamp to 1 ms so a same-timestamp fix only passes if it hasn't moved.
    const dtS = Math.max(fix.ts - prev.ts, 1) / 1000;
    computedKmh = (distanceFromPrevM(prev, fix.lat, fix.lng) / dtS) * 3.6;
    if (computedKmh > MAX_SPEED_KMH) return { accepted: false, reason: 'speed' };
  }
  return { accepted: true, bucket: speedBucket(reportedSpeedKmh(fix) ?? computedKmh ?? 0) };
}
```

```ts
import { gridDisk, gridPathCells } from 'h3-js';
import { GAP_MAX_M, GAP_MAX_MS, REVEAL_RADIUS } from './constants';

/** Larger gaps are treated as jumps (flights, tracking outages) and not filled. */
export function shouldFillGap(distanceM: number, dtMs: number): boolean {
  return distanceM < GAP_MAX_M && dtMs < GAP_MAX_MS;
}

/** Cells from `from` to `to` inclusive when filling, otherwise just `to`. */
export function pathCells(from: string | null, to: string, fill: boolean): string[] {
  if (from === null || !fill || from === to) return [to];
  try {
    return gridPathCells(from, to);
  } catch {
    // gridPathCells fails near pentagons or across some icosahedron faces;
    // treat that like a jump rather than losing the fix.
    return [to];
  }
}

export function expandCells(cells: Iterable<string>, radius: number = REVEAL_RADIUS): Set<string> {
  const out = new Set<string>();
  for (const cell of cells) {
    for (const c of gridDisk(cell, radius)) out.add(c);
  }
  return out;
}
```

```ts
import { latLngToCell } from 'h3-js';
import { slowerBucket } from './buckets';
import { H3_RES } from './constants';
import { evaluateFix } from './filter';
import { distanceFromPrevM } from './geo';
import { expandCells, pathCells, shouldFillGap } from './reveal';
import type { Fix, LastPoint, LogRow, SpeedBucket } from './types';

export type PipelineResult = {
  cells: Map<string, SpeedBucket>;
  logRows: LogRow[];
  nextLastPoint: LastPoint | null;
};

/**
 * Turns a batch of fixes into revealed cells and diagnostics. Shared by live
 * tracking and GPX import so both reveal exactly the same cells.
 */
export function processFixes(fixes: readonly Fix[], lastPoint: LastPoint | null): PipelineResult {
  const cells = new Map<string, SpeedBucket>();
  const logRows: LogRow[] = [];
  let prev = lastPoint;

  for (const fix of [...fixes].sort((x, y) => x.ts - y.ts)) {
    const verdict = evaluateFix(fix, prev);
    logRows.push({
      ts: fix.ts,
      accuracy: fix.accuracy,
      speed: fix.speed !== null && fix.speed >= 0 ? fix.speed : null,
      accepted: verdict.accepted,
      rejectReason: verdict.accepted ? null : verdict.reason,
    });
    if (!verdict.accepted) continue;

    const cell = latLngToCell(fix.lat, fix.lng, H3_RES);
    const fill =
      prev !== null && shouldFillGap(distanceFromPrevM(prev, fix.lat, fix.lng), fix.ts - prev.ts);
    for (const c of expandCells(pathCells(prev?.cell ?? null, cell, fill))) {
      const existing = cells.get(c);
      cells.set(c, existing ? slowerBucket(existing, verdict.bucket) : verdict.bucket);
    }
    prev = { cell, ts: fix.ts, lat: fix.lat, lng: fix.lng };
  }

  return { cells, logRows, nextLastPoint: prev };
}

/** The only form of a LastPoint that may be written to disk. */
export function persistable(point: LastPoint): { cell: string; ts: number } {
  return { cell: point.cell, ts: point.ts };
}
```

```ts
import { UNITS, cellArea } from 'h3-js';
import type { TrackingMode } from './types';

export function areaKm2(cells: Iterable<string>): number {
  let total = 0;
  for (const c of cells) total += cellArea(c, UNITS.km2);
  return total;
}

export function startOfLocalDay(nowMs: number): number {
  const d = new Date(nowMs);
  d.setHours(0, 0, 0, 0);
  return d.getTime();
}

/** Weeks start on Monday. */
export function startOfLocalWeek(nowMs: number): number {
  const d = new Date(startOfLocalDay(nowMs));
  const daysSinceMonday = (d.getDay() + 6) % 7;
  d.setDate(d.getDate() - daysSinceMonday);
  return d.getTime();
}

/** level is 0–1; a negative level means the platform could not read it. */
export type BatterySample = { ts: number; level: number; mode: TrackingMode };

/** Average battery drain in % per hour for each tracking mode, or null with no data. */
export function batteryRatePerHour(
  samples: readonly BatterySample[],
): Record<TrackingMode, number | null> {
  const drop: Record<TrackingMode, number> = { efficient: 0, precise: 0 };
  const hours: Record<TrackingMode, number> = { efficient: 0, precise: 0 };
  const valid = samples.filter((s) => s.level >= 0).sort((a, b) => a.ts - b.ts);

  for (let i = 1; i < valid.length; i++) {
    const a = valid[i - 1]!;
    const b = valid[i]!;
    // Skip mode switches and charging; neither says anything about drain.
    if (a.mode !== b.mode || b.level > a.level) continue;
    drop[a.mode] += (a.level - b.level) * 100;
    hours[a.mode] += (b.ts - a.ts) / 3_600_000;
  }

  const rate = (m: TrackingMode) => (hours[m] > 0 ? drop[m] / hours[m] : null);
  return { efficient: rate('efficient'), precise: rate('precise') };
}
```

```ts
import { cellsToMultiPolygon } from 'h3-js';
import type { Feature, MultiPolygon, Position } from 'geojson';

// Web Mercator ends at ~85.05°, so this covers everything MapLibre can show.
export const WORLD_RING: Position[] = [
  [-180, -85],
  [180, -85],
  [180, 85],
  [-180, 85],
  [-180, -85],
];

/**
 * Fog as one world polygon with every revealed area as a hole. Unrevealed islands
 * inside a revealed area become separate polygons, since holes cannot nest.
 */
export function buildFog(cells: string[]): Feature<MultiPolygon> {
  const outerRings: Position[][] = [];
  const islands: Position[][][] = [];
  for (const [outer, ...inner] of cellsToMultiPolygon(cells, true)) {
    if (outer) outerRings.push(outer);
    for (const ring of inner) islands.push([ring]);
  }
  return {
    type: 'Feature',
    properties: {},
    geometry: { type: 'MultiPolygon', coordinates: [[WORLD_RING, ...outerRings], ...islands] },
  };
}
```

```ts
import { distanceM } from './geo';
import type { Fix } from './types';

const POINT_RE = /<(trkpt|rtept)\b([^>]*?)(?:\/>|>([\s\S]*?)<\/\1>)/g;
const TIME_RE = /<time>([^<]+)<\/time>/;
const GPX_ACCURACY_M = 5;
const WALK_MPS = 5 / 3.6;

function numericAttr(attrs: string, name: string): number {
  const m = new RegExp(`\\b${name}\\s*=\\s*["']([^"']+)["']`).exec(attrs);
  return m ? Number(m[1]) : NaN;
}

/**
 * Dev-only GPX reader. Uses <time> when every point has one; otherwise spaces
 * points at walking pace from startMs so gap filling behaves like a walk.
 */
export function parseGpx(xml: string, startMs: number): Fix[] {
  const points: Array<{ lat: number; lng: number; time: number }> = [];
  for (const m of xml.matchAll(POINT_RE)) {
    const attrs = m[2] ?? '';
    const lat = numericAttr(attrs, 'lat');
    const lng = numericAttr(attrs, 'lon');
    if (!Number.isFinite(lat) || !Number.isFinite(lng) || Math.abs(lat) > 90 || Math.abs(lng) > 180) {
      continue;
    }
    const timeText = TIME_RE.exec(m[3] ?? '')?.[1];
    points.push({ lat, lng, time: timeText ? Date.parse(timeText) : NaN });
  }

  const hasTimes = points.length > 0 && points.every((p) => Number.isFinite(p.time));
  let synthetic = startMs;
  return points.map((p, i) => {
    const q = points[i - 1];
    if (!hasTimes && q) synthetic += (distanceM(q.lat, q.lng, p.lat, p.lng) / WALK_MPS) * 1000;
    return {
      ts: hasTimes ? p.time : Math.round(synthetic),
      lat: p.lat,
      lng: p.lng,
      accuracy: GPX_ACCURACY_M,
      speed: null,
    };
  });
}
```

```ts
import { latLngToCell } from 'h3-js';
import { H3_RES } from './constants';
import { expandCells } from './reveal';

const STEP_M = 100;
const M_PER_DEG_LAT = 111_320;
const MAX_TURN_RAD = 0.3;

/** Small seeded PRNG (mulberry32) so stress data is reproducible. */
function seededRandom(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

/** Road-like revealed cells for rendering benchmarks (Debug → stress fill). */
export function stressCells(count: number, seed: number, origin: { lat: number; lng: number }): string[] {
  const random = seededRandom(seed);
  const cells = new Set<string>();
  let { lat, lng } = origin;
  let heading = random() * 2 * Math.PI;

  while (cells.size < count) {
    heading += (random() * 2 - 1) * MAX_TURN_RAD;
    lat += (Math.cos(heading) * STEP_M) / M_PER_DEG_LAT;
    lng += (Math.sin(heading) * STEP_M) / (M_PER_DEG_LAT * Math.cos((lat * Math.PI) / 180));
    for (const c of expandCells([latLngToCell(lat, lng, H3_RES)])) cells.add(c);
  }
  return [...cells].slice(0, count);
}
```
